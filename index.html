// --- BẢO VỆ NÂNG CAO: TẢI ẢNH DẠNG BLOB ĐỂ DẤU LINK TRONG NETWORK --- const gallery = document.getElementById('gallery-container'); const fragment = document.createDocumentFragment(); const lightbox = document.getElementById('lightbox'); const lightboxImg = document.getElementById('lightbox-img'); const closeBtn = document.getElementById('close-btn'); // Lưu trữ các URL Blob đã mã hóa const blobCache = {}; async function fetchAndCreateBlobUrl(imageSrc) { if (blobCache[imageSrc]) return blobCache[imageSrc]; try { const response = await fetch(imageSrc); if (!response.ok) return null; const blob = await response.blob(); const blobUrl = URL.createObjectURL(blob); blobCache[imageSrc] = blobUrl; return blobUrl; } catch (e) { return null; } } for (let i = 1; i <= 200; i++) { const div = document.createElement('div'); div.className = 'image-container'; const fileName = i + '.png'; const img = document.createElement('img'); img.alt = 'Bảng ' + i; img.loading = 'lazy'; // Tải ảnh qua Fetch API -> Blob URL fetchAndCreateBlobUrl(fileName).then(blobUrl => { if (blobUrl) { img.src = blobUrl; } else { div.style.display = 'none'; } }); div.onclick = function() { if (blobCache[fileName]) { openLightbox(blobCache[fileName]); } }; const overlay = document.createElement('div'); overlay.className = 'image-overlay'; const numberSpan = document.createElement('span'); numberSpan.className = 'image-number'; numberSpan.innerText = i; div.appendChild(img); div.appendChild(overlay); div.appendChild(numberSpan); fragment.appendChild(div); } gallery.appendChild(fragment); function openLightbox(blobUrl) { lightboxImg.src = blobUrl; lightbox.classList.add('active'); document.body.style.overflow = 'hidden'; } function closeLightbox() { lightbox.classList.remove('active'); document.body.style.overflow = ''; setTimeout(() => { lightboxImg.src = ''; }, 300); } closeBtn.onclick = closeLightbox; lightbox.onclick = function(e) { if (e.target !== lightboxImg) { closeLightbox(); } }; document.addEventListener('keydown', function(e) { if (e.key === 'Escape' && lightbox.classList.contains('active')) { closeLightbox(); } });
